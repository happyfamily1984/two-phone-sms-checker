    private data class Item(val title: String, val from: String, val text: String, val time: String, val icon: Int, val sel: Boolean = false)

    private fun base(screen: Int, title: String, sub: String, back: Boolean, art: Int = R.drawable.header_phones): View {
        val root = paparazzi.inflate<View>(R.layout.activity_main)
        for (id in intArrayOf(R.id.screenChooser, R.id.screenCode, R.id.screenSender, R.id.screenReceiver, R.id.screenCompose, R.id.screenApps))
            root.findViewById<View>(id).visibility = if (id == screen) View.VISIBLE else View.GONE
        root.findViewById<TextView>(R.id.tvBar).text = title
        root.findViewById<TextView>(R.id.tvSub).text = sub
        root.findViewById<View>(R.id.btnBack).visibility = if (back) View.VISIBLE else View.GONE
        root.findViewById<View>(R.id.titleBadge).visibility = if (back) View.GONE else View.VISIBLE
        root.findViewById<ImageView>(R.id.ivHeaderArt).setImageResource(art)
        if (!BuildConfig.SMS_SEND) {
            root.findViewById<View>(R.id.cardSms).visibility = View.GONE
            root.findViewById<View>(R.id.btnCompose).visibility = View.GONE
        }
        val h = root.findViewById<View>(R.id.headerText)
        val d = root.resources.displayMetrics.density
        h.setPadding(h.paddingLeft, h.paddingTop + (30 * d).toInt(), h.paddingRight, h.paddingBottom)
        return root
    }

    private fun shoot(root: View, name: String) {
        val frame = FrameLayout(root.context)
        frame.addView(root, FrameLayout.LayoutParams(FrameLayout.LayoutParams.MATCH_PARENT, 2400))
        root.translationY = 0f
        paparazzi.snapshot(frame, name + "_a")
        root.translationY = -1200f
        paparazzi.snapshot(frame, name + "_b")
    }

    private fun list(root: View, items: List<Item>, selecting: Boolean) {
        val lv = root.findViewById<ListView>(R.id.lvMsgs)
        root.findViewById<View>(R.id.tvEmpty).visibility = View.GONE
        lv.adapter = object : BaseAdapter() {
            override fun getCount() = items.size
            override fun getItem(p: Int) = items[p]
            override fun getItemId(p: Int) = p.toLong()
            override fun getView(p: Int, c: View?, parent: ViewGroup): View {
                val r = LayoutInflater.from(parent.context).inflate(R.layout.item_msg, parent, false)
                val m = items[p]
                r.findViewById<TextView>(R.id.tvTitle).text = m.title
                r.findViewById<TextView>(R.id.tvFrom).text = m.from
                r.findViewById<TextView>(R.id.tvText).text = m.text
                r.findViewById<TextView>(R.id.tvTime).text = m.time
                r.findViewById<View>(R.id.card).setBackgroundResource(if (m.sel) R.drawable.card_selected else R.drawable.card_ripple)
                r.findViewById<View>(R.id.kindCircle).setBackgroundResource(if (m.sel) R.drawable.circle_primary else R.drawable.icon_circle)
                r.findViewById<ImageView>(R.id.ivKind).apply {
                    setImageResource(if (m.sel) R.drawable.ic_check else m.icon)
                    imageTintList = ColorStateList.valueOf(color(r, if (m.sel) R.color.white else R.color.primary))
                }
                r.findViewById<View>(R.id.ivChevron).visibility = if (selecting) View.INVISIBLE else View.VISIBLE
                return r
            }
        }
    }

    @Test fun compose() {
        val root = base(R.id.screenCompose, "문자 보내기", "어느 폰의 번호로 보낼까요?", true, R.drawable.header_plane)
        root.findViewById<TextView>(R.id.tvPickDevice).text = "개인폰"
        root.findViewById<EditText>(R.id.etTo).setText("010-2345-6789")
        root.findViewById<View>(R.id.btnClearTo).visibility = View.VISIBLE
        root.findViewById<EditText>(R.id.etBody).setText("네, 10시 회의 확인했습니다. 감사합니다!")
        root.findViewById<TextView>(R.id.tvCount).text = "22자"
        val sb = SpannableStringBuilder()
        fun strong(s: String) { val st = sb.length; sb.append(s)
            sb.setSpan(ForegroundColorSpan(color(root, R.color.primary)), st, sb.length, 0)
            sb.setSpan(StyleSpan(Typeface.BOLD), st, sb.length, 0) }
        sb.append("문자는 선택한 폰에서 그 폰의 번호로 발신되고, 결과는 알림으로 돌아옵니다. 그 폰에서 ")
        strong("‘받는 폰에서 문자 발신 허용’"); sb.append("이 켜져 있어야 합니다. 긴 문자(")
        strong("한글 약 70자 초과"); sb.append(")는 통신사에 따라 전송되지 않을 수 있습니다.")
        root.findViewById<TextView>(R.id.tvComposeInfo).text = sb
        shoot(root, "2_compose")
    }
